            └─ Status mudou?
               └─ notificationService mostra toast
                  └─ [Toast aparece na tela] 🎉

2. PARCEIRO RECEBE NOVO BOOKING
   └─ Cliente fez booking
      └─ usePartnerDashboardPolling detecta (10s)
         └─ Novo booking encontrado!
            └─ notificationService mostra toast
               └─ 🔔 "Novo booking!"
                  └─ Parceiro notificado imediatamente


💻 CÓDIGO EXEMPLO
═══════════════════════════════════════════════════════════════════════════════

// Antes (sem polling):
const [booking, setBooking] = useState(null);
useEffect(() => fetchBooking(), []); // Uma vez
// Nunca atualiza! Usuário precisa F5

// Depois (com polling):
const [booking, setBooking] = useState(null);
useBookingPolling({
  bookingId: id,
  interval: 5000,
  onUpdate: setBooking,
  onStatusChange: (newStatus) => 
    notificationService.notifyClientBookingUpdate('', newStatus)
});
// Atualiza a cada 5s automaticamente! Toast aparece!
